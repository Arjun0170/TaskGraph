cmake_minimum_required(VERSION 3.16)
project(taskgraph LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(ENABLE_SANITIZERS "Enable address and undefined-behavior checks" OFF)
add_library(taskgraph_core src/taskgraph.cpp)
target_include_directories(taskgraph_core PUBLIC include)
target_compile_options(taskgraph_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
if(ENABLE_SANITIZERS AND CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(taskgraph_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(taskgraph_core PUBLIC -fsanitize=address,undefined)
endif()
add_executable(taskgraph src/main.cpp)
target_link_libraries(taskgraph PRIVATE taskgraph_core)
target_compile_options(taskgraph PRIVATE -Wall -Wextra -Wpedantic -Werror)
enable_testing()
add_executable(taskgraph_tests tests/test_taskgraph.cpp)
target_link_libraries(taskgraph_tests PRIVATE taskgraph_core)
target_compile_options(taskgraph_tests PRIVATE -Wall -Wextra -Wpedantic -Werror -Wno-missing-field-initializers)
add_test(NAME taskgraph_tests COMMAND taskgraph_tests)
